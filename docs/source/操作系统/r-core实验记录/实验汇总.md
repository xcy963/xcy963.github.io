```rust
use std::collections::{BTreeMap, BinaryHeap};
use std::cmp::Reverse;

fn demo_btreemap() {
    println!("== BTreeMap demo: inode -> nlink ==");

    let mut link_counts: BTreeMap<u32, u32> = BTreeMap::new();

    // 初始化：不存在则插入 1
    link_counts.entry(10).or_insert(1);
    link_counts.entry(3).or_insert(1);

    // inc：有则 +1；无则先插入 0 再 +1（更符合“第一次变成 1”）
    *link_counts.entry(10).or_insert(0) += 1; // 10: 2
    *link_counts.entry(10).or_insert(0) += 1; // 10: 3
    *link_counts.entry(8).or_insert(0) += 1;  // 8: 1

    // get
    println!("nlink(10) = {:?}", link_counts.get(&10));
    println!("nlink(8)  = {:?}", link_counts.get(&8));

    // 有序遍历（key 从小到大）
    println!("All entries in order:");
    for (inode, nlink) in link_counts.iter() {
        println!("  inode={inode}, nlink={nlink}");
    }

    // 范围遍历（BTreeMap 很适合这个）
    println!("Range [5..=10]:");
    for (inode, nlink) in link_counts.range(5..=10) {
        println!("  inode={inode}, nlink={nlink}");
    }

    // 删除
    link_counts.remove(&3);
    println!("After remove inode=3: {:?}", link_counts);
}

fn demo_binaryheap() {
    println!("\n== BinaryHeap demo: scheduler ==");

    // 1) 默认最大堆：pop() 弹出最大
    let mut max_heap: BinaryHeap<i32> = BinaryHeap::new();
    max_heap.push(2);
    max_heap.push(10);
    max_heap.push(5);
    println!("max_heap.peek() = {:?}", max_heap.peek()); // Some(10)
    while let Some(x) = max_heap.pop() {
        println!("max_heap.pop() -> {x}");
    }

    // 2) 最小堆：用 Reverse 包起来
    // 模拟“定时任务”：按 time 最早的先执行
    let mut min_heap: BinaryHeap<Reverse<(u64, &'static str)>> = BinaryHeap::new();
    min_heap.push(Reverse((30, "flush log")));
    min_heap.push(Reverse((10, "wake task A")));
    min_heap.push(Reverse((20, "wake task B")));

    println!("\nmin-heap scheduler order:");
    while let Some(Reverse((time, name))) = min_heap.pop() {
        println!("time={time:>2} -> {name}");
    }
}

fn main() {
    demo_btreemap();
    demo_binaryheap();
}


```
