### tss
- 简而言之就是保存寄存器状态的一个数据结构,在进入task0之前
```s
pushfl
andl $0xffffbfff, (%esp)
popfl
movl $TSS0_SEL, %eax
ltr %ax				#这个其实是把内核的状态存到TSS0里面,ps:这个选择子在GDT里面
movl $LDT0_SEL, %eax
lldt %ax 
movl $0, current  #把ldt的表指向task1
sti       #本身执行上面的代码的时候计时器一直在中断，但是我们都忽略他了
pushl $0x17
pushl $init_stack
pushfl
pushl $0x0f   
pushl $task0  #执行iret返回应该会跳转到这个地址
iret  
```

- 保存内核的状态,然后task0 跳转的时候直接复活内核,这个是cpu自动的

### 问题又来了!内核进去之后从来都没有提到过e4c啊,他是哪里来的?

- 问题就在tss里面,先来看下这个数据结构
```s
tss0:	.long 0 			/* back link */
	.long krn_stk0, 0x10		/* esp0, ss0 */
	.long 0, 0, 0, 0, 0		/* esp1, ss1, esp2, ss2, cr3 */
	.long 0, 0, 0, 0, 0		/* eip, eflags, eax, ecx, edx */
	.long 0, 0, 0, 0, 0		/* ebx esp, ebp, esi, edi */
	.long 0, 0, 0, 0, 0, 0 		/* es, cs, ss, ds, fs, gs */
	.long LDT0_SEL, 0x8000000	/* ldt, trace bitmap */

	.fill 128,4,0
krn_stk0:

```

- 根据我缜密的思考,我们的代码似乎只是用到前几个字节也就是`.long krn_stk0, 0x10`,记载内核的ss和esp
